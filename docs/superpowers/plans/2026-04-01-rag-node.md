> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `rag` canvas node that connects RAG functionality (embedder, vector storage, documents) to graph agents via edges, with code generation for both context injection and tool-based access modes.

**Architecture:** New `rag` node type on the main canvas (same level as `tool`, `mcp-server`). Connected to `graph-agent` via edges. Properties panel with 4 tabs (Embedder, Storage, Documents, Mode). Backend generates Koog RAG setup code, context injection prompt, and/or `@Tool`-annotated search function.

**Tech Stack:** React/TypeScript (Rete.js), Kotlin/Spring Boot, Koog 0.7.2 RAG API

```kotlin
package com.agentwork.modeler.export.generator

import com.agentwork.modeler.export.model.KoogRagNode
import org.springframework.stereotype.Component

@Component
class RagCodeGenerator {

    fun generateSetup(rag: KoogRagNode): String {
        val varPrefix = sanitizeName(rag.name)
        return buildString {
            appendLine("// RAG: ${rag.name}")
            appendLine("val ${varPrefix}Embedder = LLMEmbedder(")
            when (rag.embedderProvider) {
                "OpenAI" -> {
                    appendLine("    OpenAILLMClient(System.getenv(\"${rag.apiKeyVariable}\")),")
                    appendLine("    OpenAIModels.Embeddings.${rag.embedderModel}")
                }
                "Ollama" -> {
                    appendLine("    OllamaClient(),")
                    appendLine("    OllamaModels.Embeddings.${rag.embedderModel}")
                }
            }
            appendLine(")")
            appendLine("val ${varPrefix}DocumentEmbedder = JVMTextDocumentEmbedder(${varPrefix}Embedder)")
            val storageExpr = when (rag.storageType) {
                "File" -> "JVMFileVectorStorage(root = Path.of(\"${rag.storagePath ?: "./vector-storage"}\"))"
                "Custom" -> "${rag.customStorageClass ?: "InMemoryVectorStorage"}()"
                else -> "InMemoryVectorStorage()"
            }
            appendLine("val ${varPrefix}Storage = EmbeddingBasedDocumentStorage(")
            appendLine("    ${varPrefix}DocumentEmbedder, $storageExpr")
            appendLine(")")
            appendLine()
            for (fp in rag.filePaths) {
                appendLine("${varPrefix}Storage.store(Path.of(\"$fp\"))")
            }
            for (url in rag.urlSources) {
                appendLine("${varPrefix}Storage.store(downloadToTempFile(\"$url\"))")
            }
        }
    }

    fun generateContextInjection(rag: KoogRagNode): String {
        val varPrefix = sanitizeName(rag.name)
        return buildString {
            appendLine("val ${varPrefix}Context = ${varPrefix}Storage")
            appendLine("    .mostRelevantDocuments(query, count = ${rag.maxDocuments}, similarityThreshold = ${rag.similarityThreshold})")
            appendLine()
            appendLine("prompt(\"rag-context-${varPrefix}\") {")
            appendLine("    user {")
            appendLine("        +\"Relevant context:\"")
            appendLine("        ${varPrefix}Context.forEach { file(it.pathString, \"text/plain\") }")
            appendLine("    }")
            appendLine("}")
        }
    }

    fun generateToolRegistration(rag: KoogRagNode): String {
        val varPrefix = sanitizeName(rag.name)
        val desc = rag.toolDescription ?: "Search for relevant documents"
        return buildString {
            appendLine("@Tool")
            appendLine("@LLMDescription(\"${desc.replace("\"", "\\\"")}\")")
            appendLine("suspend fun search${rag.name.replaceFirstChar { it.uppercase() }}(")
            appendLine("    @LLMDescription(\"Query to search relevant documents about\") query: String,")
            appendLine("    @LLMDescription(\"Maximum number of documents\") count: Int")
            appendLine("): String {")
            appendLine("    val docs = ${varPrefix}Storage")
            appendLine("        .mostRelevantDocuments(query, count = count, similarityThreshold = ${rag.similarityThreshold})")
            appendLine("        .toList()")
            appendLine("    if (docs.isEmpty()) return \"No relevant documents found for: \$query\"")
            appendLine("    return buildString {")
            appendLine("        docs.forEachIndexed { i, doc ->")
            appendLine("            append(\"Document \${i + 1}: \${doc.fileName}\\n\")")
            appendLine("            append(\"Content: \${java.nio.file.Files.readString(doc)}\\n\\n\")")
            appendLine("        }")
            appendLine("    }")
            appendLine("}")
        }
    }

    fun generateImports(ragNodes: List<KoogRagNode>): String {
        if (ragNodes.isEmpty()) return ""
        val hasOpenAI = ragNodes.any { it.embedderProvider == "OpenAI" }
        val hasOllama = ragNodes.any { it.embedderProvider == "Ollama" }
        val hasFileStorage = ragNodes.any { it.storageType == "File" }
        val hasTool = ragNodes.any { it.provideAsTool }
        val hasUrls = ragNodes.any { it.urlSources.isNotEmpty() }
        return buildString {
            appendLine("import ai.koog.prompt.embedding.LLMEmbedder")
            appendLine("import ai.koog.rag.base.embedding.JVMTextDocumentEmbedder")
            appendLine("import ai.koog.rag.base.storage.EmbeddingBasedDocumentStorage")
            appendLine("import ai.koog.rag.base.storage.InMemoryVectorStorage")
            if (hasFileStorage) appendLine("import ai.koog.rag.base.files.JVMFileVectorStorage")
            if (hasOpenAI) {
                appendLine("import ai.koog.prompt.llm.OpenAILLMClient")
                appendLine("import ai.koog.prompt.llm.OpenAIModels")
            }
            if (hasOllama) {
                appendLine("import ai.koog.prompt.llm.OllamaClient")
                appendLine("import ai.koog.prompt.llm.OllamaModels")
            }
            if (hasTool) {
                appendLine("import ai.koog.agents.core.tools.Tool")
                appendLine("import ai.koog.agents.core.tools.LLMDescription")
            }
            appendLine("import java.nio.file.Path")
            if (hasUrls) appendLine("import java.net.URL")
        }
    }

    private fun sanitizeName(name: String): String {
        return name.replaceFirstChar { it.lowercase() }
            .replace(Regex("[^a-zA-Z0-9]"), "")
    }
}
```

    <div>
      <Label className="text-xs text-muted-foreground">{label}</Label>
      <div className="space-y-1 mt-1">
        {items.map((item, i) => (
          <div key={i} className="flex gap-1 items-center">

              value={item}
              onChange={(e) => {
                const updated = [...items];
                updated[i] = e.target.value;
                onUpdate(updated);
              }}
              placeholder={placeholder}
              className="text-xs"
            />
            <Button variant="ghost" size="sm" onClick={() => onUpdate(items.filter((_, j) => j !== i))} className="text-xs px-2">x</Button>
